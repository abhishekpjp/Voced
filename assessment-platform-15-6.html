📊 Connect to Google Sheets

Every student result will be saved directly to your Google Sheet automatically — no downloads needed. Follow these one-time steps:

Step 1: Go to sheets.google.com → Create a new blank sheet
Step 2: Click Extensions → Apps Script
Step 3: Delete all existing code. Paste the code shown below → Click 💾 Save
Step 4: Click Deploy → New Deployment → Web App
Step 5: Set "Who has access" = Anyone → Click Deploy → Copy the Web App URL
Step 6: Paste that URL below → Click ✅ Save URL 
function corsResponse(text) {
  return ContentService.createTextOutput(text)
    .setMimeType(ContentService.MimeType.TEXT);
}

function handleRequest(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var p = e.parameter || {};

  if (p.action === 'saveResult') {
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(['Name','Roll No','Class','Section','Topic','Score','Total','Percentage','Grade','Result','Date','Time(mins)']);
    }
    sheet.appendRow([
      p.name || '', p.roll || '', p.cls || '', p.section || '',
      p.topic || '', p.score || '', p.total || '', p.pct || '',
      p.grade || '', p.result || '', p.date || '', p.time || ''
    ]);
    return corsResponse('OK');
  }

  if (p.action === 'getResults') {
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return corsResponse('[]');
    return ContentService.createTextOutput(JSON.stringify(data.slice(1)))
      .setMimeType(ContentService.MimeType.JSON);
  }

  return corsResponse('OK');
}

function doGet(e)  { return handleRequest(e); }
function doPost(e) { return handleRequest(e); }
